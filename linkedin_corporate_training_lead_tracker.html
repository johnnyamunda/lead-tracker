function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return ContentService.createTextOutput(JSON.stringify({status:'success', data:[]})).setMimeType(ContentService.MimeType.JSON);
  var headers = data[0], rows = [];
  for (var i = 1; i < data.length; i++) {
    var r = {}; for (var j = 0; j < headers.length; j++) r[headers[j]] = data[i][j];
    rows.push(r);
  }
  return ContentService.createTextOutput(JSON.stringify({status:'success', data:rows})).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var payload = JSON.parse(e.postData.contents);
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    
    if (payload.action === 'sync_all') {
      sheet.clearContents();
      var headers = ['id','title','fullName','firstName','lastName','position','company','source','linkedinUrl','targetProgrammes','relevancyScore','status','initialDate','dueDate','responseReceived','nextStep','responseNotes'];
      sheet.appendRow(headers);
      var list = payload.leads || [];
      for (var i = 0; i < list.length; i++) {
        var l = list[i];
        sheet.appendRow([l.id||'', l.title||'', l.fullName||'', l.firstName||'', l.lastName||'', l.position||l.role||'', l.company||'', l.source||'', l.linkedinUrl||'', Array.isArray(l.targetProgrammes)?l.targetProgrammes.join('; '):(l.targetProgrammes||''), l.relevancyScore||3, l.status||'Identified', l.initialDate||'', l.dueDate||'', l.responseReceived?true:false, l.nextStep||'', l.responseNotes||'']);
      }
      return ContentService.createTextOutput(JSON.stringify({status:'success', count:list.length})).setMimeType(ContentService.MimeType.JSON);
    }
    
    if (payload.action === 'send_digest') {
      var recipient = payload.recipientEmail;
      var dueLeads = payload.dueLeads || [];
      if (!recipient) return ContentService.createTextOutput(JSON.stringify({status:'error', message:'No recipient email provided'})).setMimeType(ContentService.MimeType.JSON);
      
      var html = '
';
      html += '
⚠️ Follow-ups Due Today (' + dueLeads.length + ')
';
      html += '
Here are your LinkedIn corporate training outreach prospects requiring follow-up today:
';
      
      if (dueLeads.length === 0) {
        html += '
🎉 All caught up! No follow-ups currently due.
';
      } else {
        html += '';
        html += '';
        for (var k = 0; k < dueLeads.length; k++) {
          var ld = dueLeads[k];
          html += '';
          html += '';
          html += '';
          html += '';
          html += '';
          html += '';
        }
        html += '
ProspectCompany & PositionDue DateNext Step
' + (ld.title ? ld.title + ' ' : '') + (ld.fullName || ld.firstName) + '
LinkedIn ↗		' + (ld.company||'') + '
' + (ld.position||ld.role||'') + '		' + (ld.dueDate||'') + '' + (ld.nextStep||'Follow up') + '
';
      }
      html += '
Generated by LeadPulse Pipeline Tracker
';
      
      MailApp.sendEmail({
        to: recipient,
        subject: '📋 LinkedIn Outreach: ' + dueLeads.length + ' Follow-up(s) Due Today',
        htmlBody: html
      });
      return ContentService.createTextOutput(JSON.stringify({status:'success', message:'Email digest sent successfully'})).setMimeType(ContentService.MimeType.JSON);
    }
    
    return ContentService.createTextOutput(JSON.stringify({status:'error', message:'Invalid action'})).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status:'error', message:err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}
